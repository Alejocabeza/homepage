## 5. Language Switcher Implementation

### Location
- **Position**: Right side of fixed header nav
- **Style**: `ES | EN` text toggle (not dropdown)

### Behavior
1. Click toggles between `/es/` and `/en/` paths
2. Preserves current page path when switching
3. Visual feedback: active language highlighted
4. Cookie storage for preference persistence

### URL Mapping Examples

| Current URL | Switch to ES | Switch to EN |
|-------------|--------------|--------------|
| `/es/` | `/es/` | `/en/` |
| `/es/projects/` | `/es/projects/` | `/en/projects/` |
| `/en/` | `/es/` | `/en/` |

### Cookie Strategy
- Cookie name: `preferred_locale`
- Value: `es` or `en`
- Expiry: 1 year
- Used for: fallback detection, initial redirect

### `src/i18n/index.ts` Utility Functions

```typescript
import esTranslations from "./es.json";
import enTranslations from "./en.json";

const translations = { es: esTranslations, en: enTranslations };

export type Locale = "es" | "en";

export function useTranslations(locale: Locale = "es") {
  const t = translations[locale];
  return (key: string): string => {
    const keys = key.split(".");
    let result: any = t;
    for (const k of keys) {
      result = result?.[k];
    }
    return result ?? key;
  };
}

export function getLocaleFromUrl(url: URL): Locale {
  const [, locale] = url.pathname.split("/");
  if (locale === "en") return "en";
  return "es";
}

export function switchLocale(url: URL, newLocale: Locale): URL {
  const pathname = url.pathname;
  const [, currentLocale, ...rest] = pathname.split("/");
  
  if (currentLocale === "en") {
    // Remove /en prefix
    const newPath = "/" + rest.join("/");
    return new URL(newPath === "/" ? newLocale + "/" : newLocale + newPath, url.origin);
  } else {
    // Add /en prefix
    const newPath = "/" + rest.join("/");
    return new URL(newLocale + (newPath === "/" ? "/" : newPath), url.origin);
  }
}

export function getLocalizedPath(path: string, locale: Locale): string {
  // Remove existing locale prefix
  const cleanPath = path.replace(/^\/(es|en)/, "");
  return `/${locale}${cleanPath || "/"}`;
}
```

### Root Redirect Logic (`src/pages/index.astro`)

```astro
---
import { switchLocale } from "@/i18n";

// Check cookie or Accept-Language header
const cookieLocale = Astro.cookies.get("preferred_locale")?.value;
const acceptLanguage = Astro.request.headers.get("accept-language");

let targetLocale: "es" | "en" = "es";

if (cookieLocale === "en" || (acceptLanguage?.startsWith("en") && !cookieLocale)) {
  targetLocale = "en";
}

// Redirect to localized path
return Astro.redirect(`/${targetLocale}/`);
---
```
